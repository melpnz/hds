### 3 — партиция манифеста (программно, все 55 записей)

```
$ node -e '
const m = require("./components/manifest.json");
let errors=[], sumCaptured=0, sumNormative=0, sumUncaptured=0;
for (const c of m.components) {
  const req = new Set(c.requiredStates);
  const cap = new Set(c.capturedStates||[]);
  const norm = new Set(c.normativeStates||[]);
  const unc = new Set(c.uncapturedStates||[]);
  if (!cap.has("default")) errors.push(c.id+": default missing");
  for (const s of req) {
    const count = (cap.has(s)?1:0)+(norm.has(s)?1:0)+(unc.has(s)?1:0);
    if (count !== 1) errors.push(c.id+"."+s+": count="+count);
  }
  for (const s of [...cap, ...norm, ...unc]) if (!req.has(s)) errors.push(c.id+"."+s+": extra");
  sumCaptured+=cap.size; sumNormative+=norm.size; sumUncaptured+=unc.size;
}
console.log("errors:", errors.length, errors);
console.log(sumCaptured, sumNormative, sumUncaptured, sumCaptured+sumNormative+sumUncaptured);
'
```
